OUT=pathlib.Path('/workspace/Twstockai/股票系統M1_v0_18/rotation_research/source_diagnostics_20261023')
RAW=OUT/'raw'; RAW.mkdir(parents=True,exist_ok=True)
URLS={
 'tcc-suspend-20250813':'https://wwwc.twse.com.tw/staticFiles/news/news/tsecnews/8a8216d697fc438f01989dd5fa3502a6.pdf',
 'tcc-resume-20250814':'https://www.twse.com.tw/staticFiles/news/news/tsecnews/8a8216d697fc438f0198a36441f702b8.pdf',
 'esun-suspend-20251105':'https://investoredu.twse.com.tw/Mobile_pages/..%2FFileSystem%2FFileUpload%2Fdae59521-69e0-4856-90fc-d349fb6013eb.pdf',

 'mof-interpretation-143802':'https://ttc.mof.gov.tw/FB/FB120/143802',
}
class NoRedirect(urllib.request.HTTPRedirectHandler):
 def redirect_request(self,*args,**kwargs): return None
def get(item):
 sid,url=item
 if (RAW/(sid+'.meta.json')).exists(): raise RuntimeError('No repeat/overwrite: '+sid)
 m=dict(url=url,requested_at=datetime.datetime.now(datetime.timezone.utc).isoformat(),method='GET',retries=0,redirects_followed=False)
 try:
  request=urllib.request.Request(url,headers={'User-Agent':'M1-Source-Audit/0.5','Accept':'application/pdf,text/html,application/json'})
  try: response=urllib.request.build_opener(NoRedirect()).open(request,timeout=40)
  except urllib.error.HTTPError as e: response=e
  with response:
   body=response.read(40*1024*1024+1)
   if len(body)>40*1024*1024: raise ValueError('Size cap')
   m.update(http=response.code,final_url=response.url,content_type=response.headers.get('Content-Type'),location=response.headers.get('Location'))
  (RAW/(sid+'.response')).write_bytes(body)
  m.update(bytes=len(body),sha256=hashlib.sha256(body).hexdigest(),status='CAPTURED' if m['http']==200 else 'HTTP_FAILURE')
  if m['http']==200 and body.startswith(b'%PDF-'):
   proc=subprocess.run(['pdftotext','-layout',str(RAW/(sid+'.response')),str(RAW/(sid+'.txt'))],capture_output=True,text=True)
   txt=RAW/(sid+'.txt')
   extraction=dict(command='pdftotext -layout',version=subprocess.run(['pdftotext','-v'],capture_output=True,text=True).stderr.splitlines()[0],returncode=proc.returncode,stderr=proc.stderr,pdf_sha256=m['sha256'])
   if txt.exists(): extraction.update(text_sha256=hashlib.sha256(txt.read_bytes()).hexdigest(),pages=txt.read_text().count('\f'))
   (RAW/(sid+'.extraction.json')).write_text(json.dumps(extraction,ensure_ascii=False,indent=2)+'\n')
   m.update(derived_text_sha256=extraction.get('text_sha256'),pdf_pages=extraction.get('pages'))
 except Exception as e: m.update(status='REQUEST_FAILED',error=type(e).__name__+': '+str(e))
 m['completed_at']=datetime.datetime.now(datetime.timezone.utc).isoformat()
 (RAW/(sid+'.meta.json')).write_text(json.dumps(m,ensure_ascii=False,indent=2)+'\n')
 return sid,m
if __name__=='__main__':
 with concurrent.futures.ThreadPoolExecutor(max_workers=4) as pool:
  results=dict(pool.map(get,URLS.items()))
 (OUT/'source-manifest.json').write_text(json.dumps(results,ensure_ascii=False,indent=2)+'\n')
 print(json.dumps(results,ensure_ascii=False,indent=2))
